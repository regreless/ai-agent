
整体的架构
src/
├── .dw/                             # 智能体工作区（dw = 大伟）
│   └── skills/                      # Skill 技能文件目录
│       ├── poem-joke.skill.md       # 诗词笑话生成器
│       ├── code-review.skill.md     # 代码审查助手
│       └── doc-generator.skill.md   # 技术文档生成器
│
├── src/                             # TypeScript 源码
│   ├── agent.ts                     # 智能体核心类（最重要）
│   ├── skill-loader.ts              # Skill 文件扫描和解析
│   ├── sandbox.ts                   # 沙箱，文件操作安全隔离
│   ├── hitl.ts                      # HITL 人为参与机制
│   ├── index.ts                     # 主入口，交互式对话模式
│   ├── demo-basic.ts                # Demo1：基础 Skill 调用
│   ├── demo-search.ts               # Demo2：搜索 + 写文件
│   ├── demo-multi-agent.ts          # Demo3：多子智能体协作
│   └── tools/
│       └── tavily-search.ts         # Tavily 搜索工具封装
│
├── output/                          # 智能体生成文件的输出目录
│                                    # 自动创建，所有 AI 写的文件都在这里
│
├── dist/                            # 编译后的 JS 文件（自动生成，不要手动改）
│
├── .env                             # 环境变量（存放 API Key，不提交 Git）
├── .env.example                     # 环境变量模板（可以提交 Git）
├── .gitignore                       # Git 忽略配置
├── package.json                     # 项目配置和依赖
├── tsconfig.json                    # TypeScript 编译配置
└── tsup.config.ts                   # tsup 构建配置


# 数据流向
用户在终端输入文字
        ↓
hitl.ts 检查是否包含高风险关键词
  → 包含：终端弹出确认框，等待 y/n
  → 不包含：直接通过
        ↓
agent.ts 把用户输入 + 对话历史 + System Prompt 一起发给 大模型（Deepseek Chatpgt）
        ↓
大模型 返回回复（普通或流式）
        ↓
agent.ts 扫描回复内容
  → 找到 ```filename:xxx.md 格式：调用 sandbox.writeFile 写文件
  → 没有文件格式：直接打印内容
        ↓
输出显示在终端 / 文件出现在 output/ 目录

# 6层架构
┌─────────────────────────────────────────┐
│         第1层：用户业务入口层              │
│   接收自然语言输入 → 安全校验 → 格式预处理  │
├─────────────────────────────────────────┤
│         第2层：持久化层                   │
│   输入落盘（内存/向量数据库/关系数据库）     │
│   保证上下文连续性，这是智能体能"思考"的基础 │
├─────────────────────────────────────────┤
│         第3层：编排层（核心）              │
│   目标驱动的自动任务规划与拆解（Planning）  │
│   任务调度 + 异常回退                     │
├─────────────────────────────────────────┤
│         第4层：子智能体层                 │
│   按角色能力切片，并发执行各自的子任务       │
│   类比：主智能体是老板，子智能体是专家打工人  │
├─────────────────────────────────────────┤
│         第5层：工具 / Skill / MCP 层      │
│   搜索、文件读写、代码执行、技能调用、MCP    │
├─────────────────────────────────────────┤
│         第6层：模型层                     │
│   本地模型（Ollama / LM Studio）          │
│   云端模型（OpenAI / 豆包 / 千问）         │
└─────────────────────────────────────────┘
